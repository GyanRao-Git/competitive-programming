const int MAX = 1e5 + 5;
bool pre = false;
vector<int> sp(MAX);

void SPF(){
    for(int i = 2; i < MAX; i++)
        sp[i] = i;

    for(int i = 2; i * i < MAX; i++){
        if(sp[i] == i){
            for(int j = i * i; j < MAX; j += i){
                if(sp[j] == j)
                    sp[j] = i;
            }
        }
    }

    pre = true;
}

        SPF = smallest prime factor of N
